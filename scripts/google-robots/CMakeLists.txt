cmake_minimum_required(VERSION 3.16)
project(linvuu_google_robots LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${GOOGLE_ROBOTS_DIR}/robots.cc" OR NOT EXISTS "${ABSEIL_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Informe GOOGLE_ROBOTS_DIR e ABSEIL_DIR com os repositórios fixados.")
endif()
set(ABSL_PROPAGATE_CXX_STD ON)
add_subdirectory("${ABSEIL_DIR}" "${CMAKE_BINARY_DIR}/abseil" EXCLUDE_FROM_ALL)
add_executable(linvuu-google-robots "${GOOGLE_ROBOTS_DIR}/robots.cc" "${GOOGLE_ROBOTS_DIR}/robots_main.cc")
target_include_directories(linvuu-google-robots PRIVATE "${GOOGLE_ROBOTS_DIR}")
target_link_libraries(linvuu-google-robots PRIVATE absl::base absl::strings)
