rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /usuarios/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    match /lessons/{lessonId} {
      allow read: if true;
      allow write: if false;
    }
    // Perfil usado pelo site. Progresso e privilegios sao geridos pelo servidor.
    match /users/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow create: if request.auth != null && request.auth.uid == uid
        && request.resource.data.keys().hasAll(['nome', 'email', 'xp', 'streak', 'licoesCompletas', 'criadoEm'])
        && request.resource.data.keys().hasOnly(['nome', 'email', 'xp', 'streak', 'licoesCompletas', 'criadoEm'])
        && request.resource.data.nome is string
        && request.resource.data.nome.size() <= 100
        && request.resource.data.email is string
        && request.resource.data.email.size() <= 254
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.xp is int && request.resource.data.xp == 0
        && request.resource.data.streak is int && request.resource.data.streak == 0
        && request.resource.data.licoesCompletas is list
        && request.resource.data.licoesCompletas.size() == 0
        && request.resource.data.criadoEm == request.time;
      allow update: if request.auth != null && request.auth.uid == uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['nome'])
        && request.resource.data.nome is string
        && request.resource.data.nome.size() <= 100;
      allow delete: if false;
    }
  }
}
